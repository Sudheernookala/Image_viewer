Integration examples

  • PLM portal: iframe + postMessage. The BOM table and the 3D view are linked both ways, parts are coloured by lifecycle state, and a snapshot is sent back to the host.
  • Web component: <cad-viewer> with its JavaScript API and events.
  • Bare embed URL: what a PLM "Open in 3D" link or tab would point to.

Angular

// app.module.ts (or standalone component): allow custom elements
schemas: [CUSTOM_ELEMENTS_SCHEMA]

// index.html
<script type="module" src="https://viewer.example.com/sdk/cad-viewer.js"></script>

// part-viewer.component.html
<cad-viewer #cad source="plm" [attr.model-id]="partNumber" ui="compact"
            (cad-select)="onSelect($event.detail.part)" style="height:600px"></cad-viewer>

// part-viewer.component.ts
@ViewChild('cad') cad!: ElementRef;
ngAfterViewInit() { this.cad.nativeElement.headers = () => ({ Authorization: 'Bearer ' + this.auth.token }) }
highlight(names: string[]) { this.cad.nativeElement.viewer?.isolate(names) }

React

// once: <script type="module" src="https://viewer.example.com/sdk/cad-viewer.js">
function PartViewer({ partNumber, onSelect }) {
  const ref = useRef(null)
  useEffect(() => {
    const el = ref.current
    const h = (e) => onSelect(e.detail.part)
    el.addEventListener('cad-select', h)
    return () => el.removeEventListener('cad-select', h)
  }, [onSelect])
  return <cad-viewer ref={ref} source="plm" model-id={partNumber} ui="compact" style={{ height: 600 }} />
}

Plain JavaScript (no UI, full control)

import { mountViewer } from 'https://viewer.example.com/sdk/cad-viewer.js'
const viewer = await mountViewer(document.getElementById('box'), { ui: 'none', headers: () => ({ Authorization: token }) })
await viewer.load({ url: '/my-dms/files/4711', name: 'housing.step' })
viewer.on('select', ({ part }) => console.log(part?.name))
viewer.colorize({ 'Screw_*': '#e11d48' })

iframe (any tool that can show a web page)

<iframe id="v" src="https://viewer.example.com/embed.html?source=plm&id=PRT-1001&ui=compact"></iframe>
<script src="https://viewer.example.com/sdk/cad-viewer-host.js"></script>
<script>
  const viewer = new CadViewerFrame(document.getElementById('v'))
  viewer.on('select', ({ part }) => showDetails(part?.name))
  viewer.isolate(['Engine_*'])
